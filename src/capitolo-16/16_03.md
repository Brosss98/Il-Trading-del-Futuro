# Il codice: un EA completo

Implementa la strategia di pullback in trend del Capitolo 15: long, EMA 50/200, RSI 14 che risale sopra 45, stop a 2 ATR, target a 2R, uscita a tempo dopo 40 barre. È un **esempio didattico**, utilizzabile su qualsiasi simbolo (oro compreso), da provare solo in demo. Il codice è diviso in tre parti.

### Parte 1: parametri, avvio e chiusura

```mql5
#include <Trade\Trade.mqh>

input double InpRiskPercent = 0.5;
input double InpRR          = 2.0;
input double InpAtrMult     = 2.0;
input int    InpMaxBars     = 40;
input int    InpMaxSpread   = 30; // punti
input ulong  InpMagic       = 160516;

CTrade   trade;
int      hEma50, hEma200, hRsi, hAtr;
datetime lastBar = 0;

int OnInit()
{
   hEma50 = iMA(_Symbol, _Period, 50, 0,
                MODE_EMA, PRICE_CLOSE);
   hEma200 = iMA(_Symbol, _Period, 200,
                 0, MODE_EMA, PRICE_CLOSE);
   hRsi = iRSI(_Symbol, _Period, 14,
               PRICE_CLOSE);
   hAtr = iATR(_Symbol, _Period, 14);
   if(hEma50 == INVALID_HANDLE ||
      hEma200 == INVALID_HANDLE ||
      hRsi == INVALID_HANDLE ||
      hAtr == INVALID_HANDLE)
      return INIT_FAILED;
   trade.SetExpertMagicNumber(InpMagic);
   return INIT_SUCCEEDED;
}

void OnDeinit(const int reason)
{
   IndicatorRelease(hEma50);
   IndicatorRelease(hEma200);
   IndicatorRelease(hRsi);
   IndicatorRelease(hAtr);
}
```

### Parte 2: funzioni di supporto

```mql5
double Buf(int h, int shift)
{
   double b[1];
   if(CopyBuffer(h, 0, shift, 1, b) != 1)
      return EMPTY_VALUE;
   return b[0];
}

bool IsNewBar()
{
   datetime t = iTime(_Symbol, _Period, 0);
   if(t == lastBar)
      return false;
   lastBar = t;
   return true;
}

bool IsOurs(ulong tk)
{
   if(!PositionSelectByTicket(tk))
      return false;
   if(PositionGetString(POSITION_SYMBOL)
      != _Symbol)
      return false;
   return PositionGetInteger(POSITION_MAGIC)
          == (long)InpMagic;
}

bool HasPosition()
{
   for(int i = PositionsTotal() - 1;
       i >= 0; i--)
      if(IsOurs(PositionGetTicket(i)))
         return true;
   return false;
}

void CheckTimeExit()
{
   for(int i = PositionsTotal() - 1;
       i >= 0; i--)
   {
      ulong tk = PositionGetTicket(i);
      if(!IsOurs(tk))
         continue;
      datetime t0 = (datetime)
         PositionGetInteger(POSITION_TIME);
      int bars = iBarShift(_Symbol,
                           _Period, t0);
      if(bars >= InpMaxBars)
         trade.PositionClose(tk);
   }
}
```

### Parte 3: lotti e logica principale

```mql5
double CalcLots(double slDist)
{
   double eq = AccountInfoDouble(
      ACCOUNT_EQUITY);
   double risk = eq * InpRiskPercent / 100.0;
   double tv = SymbolInfoDouble(_Symbol,
      SYMBOL_TRADE_TICK_VALUE);
   double ts = SymbolInfoDouble(_Symbol,
      SYMBOL_TRADE_TICK_SIZE);
   double st = SymbolInfoDouble(_Symbol,
      SYMBOL_VOLUME_STEP);
   double vmin = SymbolInfoDouble(_Symbol,
      SYMBOL_VOLUME_MIN);
   double vmax = SymbolInfoDouble(_Symbol,
      SYMBOL_VOLUME_MAX);
   if(tv <= 0 || ts <= 0 || slDist <= 0)
      return 0.0;
   double perLot = slDist / ts * tv;
   double lots = risk / perLot;
   lots = MathFloor(lots / st) * st;
   if(lots < vmin)
      return 0.0;
   return MathMin(lots, vmax);
}

bool MarginOK(double lots, double price)
{
   double m = 0.0;
   if(!OrderCalcMargin(ORDER_TYPE_BUY,
         _Symbol, lots, price, m))
      return false;
   double fm = AccountInfoDouble(
      ACCOUNT_MARGIN_FREE);
   return (m > 0.0 && m <= 0.5 * fm);
}

void OnTick()
{
   CheckTimeExit();
   if(!IsNewBar() || HasPosition())
      return;
   long spr = SymbolInfoInteger(_Symbol,
      SYMBOL_SPREAD);
   if(spr > InpMaxSpread)
      return;

   double e50  = Buf(hEma50, 1);
   double e200 = Buf(hEma200, 1);
   double r1   = Buf(hRsi, 1);
   double r2   = Buf(hRsi, 2);
   double atr  = Buf(hAtr, 1);
   if(e50 == EMPTY_VALUE ||
      e200 == EMPTY_VALUE ||
      r1 == EMPTY_VALUE ||
      r2 == EMPTY_VALUE ||
      atr == EMPTY_VALUE)
      return;

   double c1 = iClose(_Symbol, _Period, 1);
   bool trend = (e50 > e200 && c1 > e200);
   bool cross = (r2 < 45.0 && r1 >= 45.0);
   if(!(trend && cross))
      return;

   double ask = SymbolInfoDouble(_Symbol,
      SYMBOL_ASK);
   double risk = InpAtrMult * atr;
   int dg = (int)SymbolInfoInteger(
      _Symbol, SYMBOL_DIGITS);
   double sl = NormalizeDouble(
      ask - risk, dg);
   double tp = NormalizeDouble(
      ask + InpRR * risk, dg);
   double lots = CalcLots(risk);
   if(lots <= 0.0)
      return;
   if(!MarginOK(lots, ask))
      return;
   if(!trade.Buy(lots, _Symbol, ask,
                 sl, tp, "pullback"))
      Print("Errore: ",
            trade.ResultRetcode(), " ",
            trade.ResultRetcodeDescription());
}
```

### Come si collega al Python del Capitolo 15

- `Buf(h, 1)` è l'ultima candela **chiusa**: corrisponde all'indice `i` del segnale in Python. `Buf(hRsi, 2)` corrisponde a `rsi.shift(1)`.
- L'ordine parte alla **prima tick della nuova barra**: equivale a `open[i+1]`.
- Lo stop a `2 × ATR` e il target a `2R` sono gli stessi parametri `k` e `rr`.
- `IsNewBar` evita più ingressi nella stessa barra; `HasPosition` impone una sola posizione alla volta.
- `CalcLots` ricava il valore monetario del rischio dal valore del tick, arrotonda per **difetto** e restituisce 0 se il lotto minimo supererebbe il rischio fissato.
- `MarginOK` controlla con `OrderCalcMargin` che il margine richiesto non superi la metà del margine libero.
